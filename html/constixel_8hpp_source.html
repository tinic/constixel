66[[nodiscard]] inline consteval double consteval_cos(double x, int32_t terms = 10) {
67 x = x - 6.283185307179586 * static_cast<int32_t>(x / 6.283185307179586); // wrap x to [0, 2π)
68 double res = 1.0;
69 double term = 1.0;
70 const double x2 = x * x;
71 for (int32_t i = 1; i < terms; ++i) {
72 term *= -x2 / ((2 * i - 1) * (2 * i));
73 res += term;
74 }
75 return res;
76}

78[[nodiscard]] inline consteval double consteval_sin(double x, int32_t terms = 10) {
79 x = x - 6.283185307179586 * static_cast<int32_t>(x / 6.283185307179586); // wrap x to [0, 2π)
80 double res = x;
81 double term = x;
82 const double x2 = x * x;
83 for (int32_t i = 1; i < terms; ++i) {
84 term *= -x2 / ((2 * i) * (2 * i + 1));
85 res += term;
86 }
87 return res;
88}

90inline constexpr float fast_sqrtf(const float x) {
91 auto i = std::bit_cast<int32_t>(x);
92 const int k = i & 0x00800000;
93 float y = 0.0f;
94 if (k != 0) {
95 i = 0x5ed9d098 - (i >> 1);
96 y = std::bit_cast<float>(i);
97 y = 2.33139729f * y * ((-x * y * y) + 1.07492042f);
98 } else {
99 i = 0x5f19d352 - (i >> 1);
100 y = std::bit_cast<float>(i);
101 y = 0.82420468f * y * ((-x * y * y) + 2.14996147f);
102 }
103 const float c = x * y;
104 const float r = ((y * -c) + 1.0f);
105 y = ((0.5f * c * r) + c);
106 return y;
107}

109[[nodiscard]] inline constexpr float fast_exp2(const float p) {
110 const float offset = (p < 0) ? 1.0f : 0.0f;
111 const float clipp = (p < -126) ? -126.0f : p;
112 const float z = clipp - static_cast<float>(static_cast<int32_t>(clipp)) + offset;
113 return std::bit_cast<float>(
114 static_cast<uint32_t>((1 << 23) * (clipp + 121.2740575f + 27.7280233f / (4.84252568f - z) - 1.49012907f * z)));
115}

117[[nodiscard]] inline constexpr float fast_log2(const float x) {
118 const auto xi = std::bit_cast<uint32_t>(x);
119 const auto xf = std::bit_cast<float>((xi & 0x007FFFFF) | 0x3f000000);
120 const float y = static_cast<float>(xi) * 1.1920928955078125e-7f;
121 return y - 124.22551499f - 1.498030302f * xf - 1.72587999f / (0.3520887068f + xf);
122}

124[[nodiscard]] inline constexpr float fast_pow(const float x, const float p) {
125 return fast_exp2(p * fast_log2(x));

128[[nodiscard]] inline constexpr float linear_to_srgb(float c) {
129 if (c <= 0.0031308f) {
130 return 12.92f * c;
131 }
132 if (std::is_constant_evaluated()) {
133 return 1.055f * fast_pow(c, 1.0f / 2.4f) - 0.055f;
134 }
135 return 1.055f * std::pow(c, 1.0f / 2.4f) - 0.055f;
136}

138[[nodiscard]] inline constexpr float srgb_to_linear(float s) {
139 if (s <= 0.040449936f) {
140 return s / 12.92f;
141 }
142 if (std::is_constant_evaluated()) {
143 return fast_pow((s + 0.055f) / 1.055f, 2.4f);
144 }
145 return std::pow((s + 0.055f) / 1.055f, 2.4f);
146}

161inline float32x4_t fast_exp2_f32_neon(float32x4_t p) {
162 float32x4_t one = vdupq_n_f32(1.0f);
163 float32x4_t zero = vdupq_n_f32(0.0f);
164 float32x4_t neg126 = vdupq_n_f32(-126.0f);
165 float32x4_t offset = vbslq_f32(vcltq_f32(p, zero), one, zero);
166 float32x4_t clipp = vmaxq_f32(p, neg126);
167 int32x4_t ipart = vcvtq_s32_f32(clipp);
168 float32x4_t fpart = vsubq_f32(clipp, vcvtq_f32_s32(ipart));
169 float32x4_t z = vaddq_f32(fpart, offset);
170 float32x4_t a = vdupq_n_f32(121.2740575f);
171 float32x4_t b = vdupq_n_f32(27.7280233f);
172 float32x4_t c = vdupq_n_f32(4.84252568f);
173 float32x4_t d = vdupq_n_f32(1.49012907f);
174 float32x4_t t = vaddq_f32(clipp, vsubq_f32(a, vmulq_f32(d, z)));
175 t = vaddq_f32(t, vdivq_f32(b, vsubq_f32(c, z)));
176 int32x4_t res = vcvtq_s32_f32(vmulq_n_f32(t, static_cast<float>(1 << 23)));
177 return vreinterpretq_f32_s32(res);
178}

198inline __m128 fast_exp2_ps(__m128 p) {
199 const __m128 one = _mm_set1_ps(1.0f);
200 const __m128 zero = _mm_setzero_ps();
201 const __m128 neg126 = _mm_set1_ps(-126.0f);
202 __m128 offset = _mm_and_ps(_mm_cmplt_ps(p, zero), one);
203 __m128 clipp = _mm_max_ps(p, neg126);
204 __m128i ipart = _mm_cvttps_epi32(clipp);
205 __m128 fpart = _mm_sub_ps(clipp, _mm_cvtepi32_ps(ipart));
206 __m128 z = _mm_add_ps(fpart, offset);
207 const __m128 c1 = _mm_set1_ps(121.2740575f);
208 const __m128 c2 = _mm_set1_ps(27.7280233f);
209 const __m128 c3 = _mm_set1_ps(4.84252568f);
210 const __m128 c4 = _mm_set1_ps(1.49012907f);
211 const __m128 bias = _mm_set1_ps(1 << 23);
212 __m128 t = _mm_add_ps(clipp, _mm_sub_ps(c1, _mm_mul_ps(c4, z)));
213 t = _mm_add_ps(t, _mm_div_ps(c2, _mm_sub_ps(c3, z)));
214 __m128i result = _mm_cvtps_epi32(_mm_mul_ps(t, bias));
215 return _mm_castsi128_ps(result);
216}

218inline __m128 fast_log2_ps(__m128 x) {
219 const __m128i xi = _mm_castps_si128(x);
220 const __m128i mant_mask = _mm_set1_epi32(0x007FFFFF);
221 const __m128i one_bits = _mm_set1_epi32(0x3f000000);
222 const __m128 y = _mm_mul_ps(_mm_cvtepi32_ps(xi), _mm_set1_ps(1.1920928955078125e-7f)); // 1/(1<<23)
223 __m128i mant_bits = _mm_or_si128(_mm_and_si128(xi, mant_mask), one_bits);
224 __m128 xf = _mm_castsi128_ps(mant_bits);
225 const __m128 c0 = _mm_set1_ps(124.22551499f);
226 const __m128 c1 = _mm_set1_ps(1.498030302f);
227 const __m128 c2 = _mm_set1_ps(1.72587999f);
228 const __m128 c3 = _mm_set1_ps(0.3520887068f);
229 return _mm_sub_ps(_mm_sub_ps(y, _mm_add_ps(c0, _mm_mul_ps(c1, xf))), _mm_div_ps(c2, _mm_add_ps(c3, xf)));
230}

240inline __m128 linear_to_srgb_approx_sse(__m128 l) {
241 const __m128 cutoff = _mm_set1_ps(0.0031308f);
242 const __m128 scale = _mm_set1_ps(12.92f);
243 const __m128 a = _mm_set1_ps(1.055f);

272 const double l = oklab.l;
273 const double a = oklab.a;
274 const double b = oklab.b;
275
276 const double l_ = l + 0.3963377774 * a + 0.2158037573 * b;
277 const double m_ = l - 0.1055613458 * a - 0.0638541728 * b;
278 const double s_ = l - 0.0894841775 * a - 1.2914855480 * b;
279
280 const double r = 4.0767416621 * l_ - 3.3077115913 * m_ + 0.2309699292 * s_;
281 const double g = -1.2684380046 * l_ + 2.6097574011 * m_ - 0.3413193965 * s_;
282 const double bl = -0.0041960863 * l_ - 0.7034186168 * m_ + 1.7076147031 * s_;

285 .g = static_cast<double>(linear_to_srgb(std::max(0.0f, std::min(1.0f, static_cast<float>(g))))),
286 .b = static_cast<double>(linear_to_srgb(std::max(0.0f, std::min(1.0f, static_cast<float>(bl)))))};
287}

378 [[nodiscard]] constexpr uint8_t nearest(int32_t r, int32_t g, int32_t b) const {
379 int32_t best = 0;
380 int32_t bestd = 1UL << 30;
381 for (size_t i = 0; i < pal.size(); ++i) {
382 const int32_t dr = r - static_cast<int32_t>((pal[i] >> 0) & 0xFF);
383 const int32_t dg = g - static_cast<int32_t>((pal[i] >> 8) & 0xFF);
384 const int32_t db = b - static_cast<int32_t>((pal[i] >> 16) & 0xFF);
385 const int32_t d = dr * dr + dg * dg + db * db;
386 if (d < bestd) {
387 bestd = d;
388 best = static_cast<int32_t>(i);
389 }
390 }
391 return static_cast<uint8_t>(best);
392 }

402 [[nodiscard]] constexpr uint8_t nearest_linear(float r, float g, float b) const {
403 const float epsilon = 0.0001f;
404#if defined(__ARM_NEON)
405 if (!std::is_constant_evaluated() && pal.size() >= 4) {
406 const float32x4_t vR = vdupq_n_f32(r);
407 const float32x4_t vG = vdupq_n_f32(g);
408 const float32x4_t vB = vdupq_n_f32(b);
409
410 float best = std::numeric_limits<float>::infinity();
411 std::size_t bestIdx = 0;
412
413 for (size_t i = 0; i < pal.size(); i += 4) {
414 const float32x4_t dr = vsubq_f32(vld1q_f32(&linearpal_neon[i * 3]), vR);
415 const float32x4_t dg = vsubq_f32(vld1q_f32(&linearpal_neon[i * 3 + 4]), vG);
416 const float32x4_t db = vsubq_f32(vld1q_f32(&linearpal_neon[i * 3 + 8]), vB);
417
418#if defined(__aarch64__) && defined(__ARM_FEATURE_FMA)
419 const float32x4_t dist = vfmaq_f32(vfmaq_f32(vmulq_f32(dr, dr), dg, dg), db, db);
420#else
421 const float32x4_t dist = vaddq_f32(vaddq_f32(vmulq_f32(dr, dr), vmulq_f32(dg, dg)), vmulq_f32(db, db));
422#endif
423
424 const float32x2_t lo = vget_low_f32(dist);
425 const float32x2_t hi = vget_high_f32(dist);
426 const float d0 = vget_lane_f32(lo, 0);
427 if (d0 < best) {
428 best = d0;
429 bestIdx = i;
430 if (d0 <= epsilon) {
431 break;
432 }
433 }
434 const float d1 = vget_lane_f32(lo, 1);
435 if (d1 < best) {
436 best = d1;
437 bestIdx = i + 1;
438 if (d1 <= epsilon) {
439 break;
440 }
441 }
442 const float d2 = vget_lane_f32(hi, 0);
443 if (d2 < best) {
444 best = d2;
445 bestIdx = i + 2;
446 if (d2 <= epsilon) {
447 break;
448 }
449 }
450 const float d3 = vget_lane_f32(hi, 1);
451 if (d3 < best) {
452 best = d3;
453 bestIdx = i + 3;
454 if (d3 <= epsilon) {
455 break;
456 }
457 }
458 }
459 return static_cast<uint8_t>(bestIdx);
460 }
461#endif // #if defined(__ARM_NEON)
462#if defined(__AVX2__)
463 if (!std::is_constant_evaluated() && pal.size() >= 8) {
464 const __m256 vR = _mm256_set1_ps(r);
465 const __m256 vG = _mm256_set1_ps(g);
466 const __m256 vB = _mm256_set1_ps(b);
467
468 float best = std::numeric_limits<float>::infinity();
469 std::uint8_t bestIdx = 0;
470
471 for (size_t i = 0; i < pal.size(); i += 8) {
472 __m256 pr = _mm256_load_ps(&linearpal_avx2[i * 3]);
473 __m256 pg = _mm256_load_ps(&linearpal_avx2[i * 3 + 8]);
474 __m256 pb = _mm256_load_ps(&linearpal_avx2[i * 3 + 16]);
475
476 __m256 dist =
477 _mm256_fmadd_ps(_mm256_sub_ps(pr, vR), _mm256_sub_ps(pr, vR),
478 _mm256_fmadd_ps(_mm256_sub_ps(pg, vG), _mm256_sub_ps(pg, vG),
479 _mm256_mul_ps(_mm256_sub_ps(pb, vB), _mm256_sub_ps(pb, vB))));
480
481 alignas(32) float d[8];
482 _mm256_store_ps(d, dist);
483
484 for (size_t lane = 0; lane < 8; lane++) {
485 if (d[lane] < best) {
486 best = d[lane];
487 bestIdx = static_cast<uint8_t>(i + lane);
488 if (d[lane] <= epsilon) {
489 return static_cast<uint8_t>(bestIdx);
490 }
491 }
492 }
493 }
494
495 return static_cast<uint8_t>(bestIdx);
496 }
497#endif // #if defined(__AVX2__)
498 size_t best = 0;
499 float bestd = 100.0f;
500 for (size_t i = 0; i < pal.size(); ++i) {
501 const float dr = r - linearpal.at(i * 3 + 0);
502 const float dg = g - linearpal.at(i * 3 + 1);
503 const float db = b - linearpal.at(i * 3 + 2);
504 const float d = dr * dr + dg * dg + db * db;
505 if (d < bestd) {
506 bestd = d;
507 best = i;
508 if (d <= epsilon) {
509 break;
510 }
511 }
512 }
513 return static_cast<uint8_t>(best);
514 }

521template <size_t N, typename T>
522class hextree {
523 static constexpr T bitslices = ((sizeof(T) * 8) / 4) - 1;
524
525 static constexpr size_t child_nodes_n = 1UL << 4;
526 static constexpr size_t child_nodes_n_mask = child_nodes_n - 1;
527
528 struct node {
529 std::array<T, child_nodes_n> child{};
530 constexpr node() {
531 for (auto &c : child) {
532 c = invalid;
533 }
534 }
535 };
536
537 public:
538 static constexpr T invalid = std::numeric_limits<T>::max();
539
540 std::array<node, N> nodes{};
541
542 [[nodiscard]] consteval size_t byte_size() const {
543 return sizeof(node) * nodes.size();
544 }
545
546 ~hextree() = default;
547 hextree(const hextree &&) = delete;
548 hextree(const hextree &) = delete;
549 hextree &operator=(const hextree &) = delete;
550 hextree &operator=(hextree &&) = delete;
551
552 explicit consteval hextree() = default;
553 template <std::size_t NS>
554 explicit consteval hextree(const std::array<std::pair<T, T>, NS> &in) {
555 nodes[0] = node{};
556 T node_cnt = 1;
557 for (auto [key, val] : in) {
558 T idx = 0;
559 for (T d = 0; d < bitslices; d++) {
560 T nib = (key >> ((bitslices - d) * 4)) & child_nodes_n_mask;
561 T next = nodes.at(idx).child[nib];
562 if (next == invalid) {
563 next = node_cnt;
564 nodes.at(idx).child[nib] = next;
565 node_cnt++;
566 }
567 idx = next;
568 }
569 nodes.at(idx).child[key & child_nodes_n_mask] = val;
570 }
571 }
572
573 template <std::size_t NS>
574 [[nodiscard]] static consteval T size(const std::array<std::pair<T, T>, NS> &in) {
575 std::vector<node> vnodes{1};
576 vnodes.assign(1, node{});
577 for (auto [key, val] : in) {
578 T idx = 0;
579 for (T d = 0; d < bitslices; d++) {
580 T nib = (key >> ((bitslices - d) * 4)) & child_nodes_n_mask;
581 T next = vnodes.at(idx).child[nib];
582 if (next == invalid) {
583 next = static_cast<T>(vnodes.size());
584 vnodes[idx].child[nib] = next;
585 vnodes.emplace_back();
586 }
587 idx = next;
588 }
589 vnodes[idx].child[key & child_nodes_n_mask] = val;
590 }
591 return static_cast<T>(vnodes.size());
592 }
593
594 [[nodiscard]] constexpr T lookup(T key) const {
595 T idx = 0;
596 for (T d = 0; d < bitslices; ++d) {
597 T nib = (key >> ((bitslices - d) * 4)) & child_nodes_n_mask;
598 T next = nodes.at(idx).child[nib];
599 if (next == invalid) {
600 return invalid;
601 }
602 idx = next;
603 }
604 return nodes.at(idx).child[key & child_nodes_n_mask];
605 }

693 [[nodiscard]] static constexpr uint32_t adler32(const uint8_t *data, std::size_t len, uint32_t adler32_sum) {
694 uint32_t adler32_s1 = adler32_sum & 0xFFFF;

698 adler32_s2 = (adler32_s2 + adler32_s1) % 65521;

700 return ((adler32_s2 << 16) | adler32_s1);

825 template <typename F>
826 [[nodiscard]] static constexpr uint32_t png_idat_zlib_stream(F &&char_out, const uint8_t *line, size_t bytes,
827 uint32_t adler32_sum, bool last_line) {
828 const auto max_data_use = size_t{1024};
829 const auto extra_data = size_t{10};
830 const size_t max_stack_use = max_data_use + extra_data;
831 std::array<uint8_t, max_stack_use> header{};
832 size_t filter_first = 1;
833 while (bytes > 0) {
834 size_t i = 0;
835 header.at(i++) = 'I';
836 header.at(i++) = 'D';
837 header.at(i++) = 'A';
838 header.at(i++) = 'T';
839
840 const size_t bytes_to_copy = std::min(max_data_use, bytes);
841
842 header.at(i++) = ((bytes_to_copy == bytes) && last_line) ? 0x01 : 0x00;
843
844 header.at(i++) = (((bytes_to_copy + filter_first) >> 0) & 0xFF);
845 header.at(i++) = (((bytes_to_copy + filter_first) >> 8) & 0xFF);
846 header.at(i++) = ((((bytes_to_copy + filter_first) ^ 0xffff) >> 0) & 0xFF);
847 header.at(i++) = ((((bytes_to_copy + filter_first) ^ 0xffff) >> 8) & 0xFF);
848
849 const size_t adlersum32_start_pos = i;
850 if (filter_first != 0) {
851 filter_first = 0;
852 header.at(i++) = 0;
853 }
854 for (size_t d = 0; d < bytes_to_copy; d++) {
855 header.at(i++) = *line++;
856 }
857 adler32_sum = adler32(&header[adlersum32_start_pos], i - adlersum32_start_pos, adler32_sum);
858
859 png_chunk(std::forward<F>(char_out), header, i);
860
861 bytes -= bytes_to_copy;
862 }
863
864 return adler32_sum;
865 }

957 template <typename PBT, size_t PBS>
958 struct palette_bitset {
959 constexpr void mark(PBT col) {
960 const PBT idx = (col >> 5) & set_idx_mask;
961 set[idx] |= 1UL << (col & 0x1F);
962 }
963
964 constexpr void clear() {
965 set.fill(0);
966 }
967
968 [[nodiscard]] constexpr size_t genstack(std::array<PBT, PBS> &stack) const {
969 size_t count = 0;
970 for (size_t c = 0; c < PBS / 32; c++) {
971 for (size_t d = 0; d < 32; d++) {
972 if ((set[c] >> d) & 1) {
973 stack.at(count++) = static_cast<PBT>(c * 32 + d);
974 }
975 }
976 }
977 return count;
978 }
979
980 std::array<uint32_t, PBS / 32> set{};
981 static constexpr size_t set_idx_mask = (1UL << sizeof(set) / 4) - 1;
982 };

995 adler32_sum, y == pix.height - 1);

998 png_end(std::forward<F>(char_out));

1001 // Single sixel emission body — shared between compile-time call
1002 // sites and runtime-sized images. W/H/S are runtime parameters
1003 // so the same code drives both paths.
1004 template <typename PBT, size_t PBS, typename P, typename F, typename C, typename D>
1005 static constexpr void sixel_image(const uint8_t *data, const P &palette, F &&char_out, const rect<int32_t> &_r,
1006 size_t W, size_t H, size_t S, const C &collect6, const D &set6) {
1007 sixel_header(std::forward<F>(char_out));
1008 sixel_raster_attributes(std::forward<F>(char_out), W, H, S);
1009 for (size_t c = 0; c < palette.size(); c++) {
1010 sixel_color(std::forward<F>(char_out), static_cast<uint16_t>(c), palette[c]);
1011 }
1012 const auto r =
1013 rect<int32_t>{.x = _r.x * static_cast<int32_t>(S),
1014 .y = _r.y * static_cast<int32_t>(S),
1015 .w = _r.w * static_cast<int32_t>(S),
1016 .h = _r.h * static_cast<int32_t>(S)} &
1017 rect<int32_t>{.x = 0, .y = 0, .w = static_cast<int32_t>(W * S), .h = static_cast<int32_t>(H * S)};
1018 std::array<PBT, PBS> stack{};
1019 palette_bitset<PBT, PBS> pset{};
1020 for (int32_t y = r.y; y < (r.y + r.h); y += 6) {
1021 pset.clear();
1022 set6(data, static_cast<size_t>(r.x), static_cast<size_t>(r.w), static_cast<size_t>(y), pset);
1023 const size_t stack_count = pset.genstack(stack);
1024 for (size_t s = 0; s < stack_count; s++) {
1025 const PBT col = stack[s];
1026 // Carriage-return-within-band ($) ONLY between layers,
1027 // never before the first. The previous `col != 0` test
1028 // assumed the built-in palette always has color 0 at
1029 // stack[0]; that's false for runtime palettes whose
1030 // index 0 is a reserved sentinel never marked into the
1031 // pset, so the wrong test emitted a stray $ at every
1032 // band start and Windows Terminal rendered black
1033 // artifacts for it.
1034 if (s != 0) {
1035 std::forward<F>(char_out)('$');
1036 }
1037 std::forward<F>(char_out)('#');
1038 sixel_number(std::forward<F>(char_out), static_cast<uint16_t>(col));
1039 for (int32_t x = r.x; x < (r.x + r.w); x++) {
1040 PBT bits6 = collect6(data, static_cast<size_t>(x), col, static_cast<size_t>(y));
1041 uint16_t repeat_count = 0;
1042 for (int32_t xr = x + 1; xr < (std::min(x + int32_t{255}, static_cast<int32_t>(W * S))); xr++) {
1043 if (bits6 == collect6(data, static_cast<size_t>(xr), col, static_cast<size_t>(y))) {
1044 repeat_count++;
1045 continue;
1046 }
1047 break;
1048 }
1049 if (repeat_count > uint16_t{3}) {
1050 std::forward<F>(char_out)('!');
1051 sixel_number(std::forward<F>(char_out), static_cast<uint16_t>(repeat_count + 1));
1052 x += repeat_count;
1053 }
1054 std::forward<F>(char_out)(static_cast<char>('?' + bits6));
1055 }
1056 }
1057 std::forward<F>(char_out)('-');
1058 }
1059 sixel_end(std::forward<F>(char_out));
1060 }

1086 [[nodiscard]] constexpr uint8_t nearest(int32_t r, int32_t g, int32_t b) const {
1087 int32_t best_d = std::numeric_limits<int32_t>::max();
1088 uint8_t best_i = 1;
1089 for (size_t i = 1; i < count_; ++i) {
1090 const int32_t dr = r - int32_t((pal[i] >> 0) & 0xFF);
1091 const int32_t dg = g - int32_t((pal[i] >> 8) & 0xFF);
1092 const int32_t db = b - int32_t((pal[i] >> 16) & 0xFF);
1093 const int32_t d = dr * dr + dg * dg + db * db;
1094 if (d < best_d) {
1095 best_d = d;
1096 best_i = uint8_t(i);
1097 if (d == 0)
1098 break;
1099 }
1100 }
1101 return best_i;
1102 }

1125 return std::array<uint32_t, 2>({{0x00000000, 0x00ffffff}});
1126 } else if constexpr (BITS == 2) {
1127 if (GRAYSCALE) {

1129 }
1130 return std::array<uint32_t, (1UL << bits_per_pixel)>({{0x000000, 0xffffff, 0xff0000, 0x0077ff}});
1131 } else if constexpr (BITS == 4) {

1135 0x999999, 0xaaaaaa, 0xbbbbbb, 0xcccccc, 0xdddddd, 0xeeeeee, 0xffffff}});

1138 {{0x000000, 0xffffff, 0xff0000, 0x00ff00, 0x0000ff, 0xffff00, 0x00ffff, 0xff00ff, 0x333333, 0x666666,
1139 0x999999, 0xcccccc, 0x7f0000, 0x007f00, 0x00007f, 0x7f7f00}});

1144 pal[c] = static_cast<uint32_t>((c << 16) | (c << 8) | c);
1145 }
1146 } else {

1148 pal[1] = 0xffffff;
1149 pal[2] = 0xff0000;
1150 pal[3] = 0x00ff00;
1151 pal[4] = 0x0000ff;

1154 pal[7] = 0xff00ff;

1157 pal[10] = 0x999999;
1158 pal[11] = 0xcccccc;
1159 pal[12] = 0x7f0000;
1160 pal[13] = 0x007f00;

1173 pal[0x30 + c + 8] = (x << 16) | (255 << 8) | (x << 0);

1178 pal[0x60 + c + 0] = (0 << 16) | (y << 8) | (y << 0);
1179 pal[0x60 + c + 8] = (x << 16) | (255 << 8) | (255 << 0);
1180 pal[0x70 + c + 0] = (y << 16) | (0 << 8) | (y << 0);
1181 pal[0x70 + c + 8] = (255 << 16) | (x << 8) | (255 << 0);

1218 static constexpr void transpose8x8(std::array<uint8_t, 8> &a) {
1219 // clang-format off
1220 uint32_t x = (static_cast<uint32_t>(a[0]) << 24) |
1221 (static_cast<uint32_t>(a[1]) << 16) |
1222 (static_cast<uint32_t>(a[2]) << 8) |
1223 (static_cast<uint32_t>(a[3]));
1224 uint32_t y = (static_cast<uint32_t>(a[4]) << 24) |
1225 (static_cast<uint32_t>(a[5]) << 16) |
1226 (static_cast<uint32_t>(a[6]) << 8) |
1227 (static_cast<uint32_t>(a[7]));
1228
1229 uint32_t t = (x ^ (x >> 7)) & 0x00AA00AA; x = x ^ t ^ (t << 7);
1230 t = (y ^ (y >> 7)) & 0x00AA00AA; y = y ^ t ^ (t << 7);
1231 t = (x ^ (x >> 14)) & 0x0000CCCC; x = x ^ t ^ (t << 14);
1232 t = (y ^ (y >> 14)) & 0x0000CCCC; y = y ^ t ^ (t << 14);
1233
1234 t = ((y >> 4) & 0x0F0F0F0F) | (x & 0xF0F0F0F0);
1235 y = ((x << 4) & 0xF0F0F0F0) | (y & 0x0F0F0F0F);
1236
1237 a[0] = static_cast<uint8_t>(t >> 24);
1238 a[1] = static_cast<uint8_t>(t >> 16);
1239 a[2] = static_cast<uint8_t>(t >> 8);
1240 a[3] = static_cast<uint8_t>(t);
1241 a[4] = static_cast<uint8_t>(y >> 24);
1242 a[5] = static_cast<uint8_t>(y >> 16);
1243 a[6] = static_cast<uint8_t>(y >> 8);
1244 a[7] = static_cast<uint8_t>(y);
1245 // clang-format on
1246 }

1248 template <bool FLIP_H = false, bool FLIP_V = false>
1249 static constexpr void transpose(const const_raster &pix, uint8_t *dst) {
1250 if constexpr (BITS == 1) {
1251 std::array<uint8_t, 8> tmp{};
1252 const size_t src_bytes = ((pix.width + 7) / 8);
1253 const size_t dst_stride = ((pix.height + 7) / 8);
1254 for (size_t y = 0; y < dst_stride; y++) {
1255 for (size_t x = 0; x < src_bytes; x++) {
1256 size_t xl = std::min(size_t{8}, pix.height - (y * 8));
1257 for (size_t c = 0; c < xl; c++) {
1258 if constexpr (FLIP_V) {
1259 tmp[c] = pix.data[(pix.height - 1 - (y * 8 + c)) * pix.stride + x];
1260 } else {
1261 tmp[c] = pix.data[(y * 8 + c) * pix.stride + x];
1262 }
1263 }
1264 for (size_t c = xl; c < 8; c++) {
1265 tmp[c] = 0;
1266 }
1267 transpose8x8(tmp);
1268 size_t yl = std::min(size_t{8}, pix.width - (x * 8));
1269 for (size_t c = 0; c < yl; c++) {
1270 if constexpr (FLIP_H) {
1271 dst[((src_bytes - x - 1) * 8 + 7 - c) * dst_stride + y] = tmp[c];
1272 } else {
1273 dst[(x * 8 + c) * dst_stride + y] = tmp[c];
1274 }
1275 }
1276 }
1277 }
1278 } else {
1279 // FLIP_H mirrors x on the 8-bit format and y on the 2 and 4-bit formats, FLIP_V the other axis.
1280 const bool flip_x = BITS == 8 ? FLIP_H : FLIP_V;
1281 const bool flip_y = BITS == 8 ? FLIP_V : FLIP_H;
1282 const raster out{dst, pix.height, pix.width, bytes_per_line(pix.height)};
1283 for (size_t y = 0; y < pix.height; y++) {
1284 for (size_t x = 0; x < pix.width; x++) {
1285 plot(out, flip_y ? (pix.height - 1 - y) : y, flip_x ? (pix.width - 1 - x) : x, get_col(pix, x, y));
1286 }
1287 }
1288 }

1295 static_assert(SUPPORTED, "composing not supported on 1-bit and 2-bit formats, use a mono font.");
1296#endif // #ifndef _MSC_VER

1319 // Note: this reads 1 float into linearpal_avx2
1320 __m128 bg_rgb = _mm_loadu_ps(&quant.linear_palette()[bg * 3]);
1321 __m128 result_rgb = _mm_add_ps(_mm_mul_ps(col_rgb, cola_v), _mm_mul_ps(bg_rgb, inv_cola_v));

1324 plot(pix, x, y, quant.nearest_linear(result[0], result[1], result[2]));
1325 } else
1326#endif // #if defined(__AVX2__)
1327 {
1328 const auto bg = static_cast<size_t>(pix.data[y * pix.stride + x]);
1329 const float Rl = colr * cola + quant.linear_palette().at(bg * 3 + 0) * (1.0f - cola);
1330 const float Gl = colg * cola + quant.linear_palette().at(bg * 3 + 1) * (1.0f - cola);
1331 const float Bl = colb * cola + quant.linear_palette().at(bg * 3 + 2) * (1.0f - cola);
1332 plot(pix, x, y, quant.nearest_linear(Rl, Gl, Bl));
1333 }

1335 } else if constexpr (BITS == 4) {

1337 const float Rl = colr * cola + quant.linear_palette().at(bg * 3 + 0) * (1.0f - cola);

1339 const float Bl = colb * cola + quant.linear_palette().at(bg * 3 + 2) * (1.0f - cola);

1355 static constexpr void extent(const raster &pix, size_t xl0, size_t xr0, size_t y, uint8_t col) {
1356 uint8_t *yptr = &pix.data[y * pix.stride];
1357 if constexpr (BITS == 8) {
1358 for (size_t x = xl0; x < xr0; x++) {
1359 yptr[x] = col;
1360 }
1361 } else {
1362 const size_t xl = xl0 / pixels_per_byte;
1363 const size_t xr = xr0 / pixels_per_byte;
1364 // col in every pixel of a byte, and the pixels of a byte right of / left of a position
1365 const auto fill = static_cast<uint8_t>((col & color_mask) * (0xFF / color_mask));
1366 const auto ml = static_cast<uint8_t>(0xFF >> ((xl0 % pixels_per_byte) * BITS));
1367 const auto mr = static_cast<uint8_t>(0xFF00 >> ((xr0 % pixels_per_byte) * BITS));
1368 if (xr > xl) {
1369 yptr[xl] = static_cast<uint8_t>((yptr[xl] & ~ml) | (ml & fill));
1370 for (size_t x = xl + 1; x < xr; x++) {
1371 yptr[x] = fill;
1372 }
1373 if (mr != 0) {
1374 yptr[xr] = static_cast<uint8_t>((yptr[xr] & ~mr) | (mr & fill));
1375 }
1376 } else {
1377 yptr[xl] = static_cast<uint8_t>((yptr[xl] & ~(ml & mr)) | (ml & mr & fill));
1378 }
1379 }
1380 }

1465 static constexpr void blit_RGBA_diffused_linear(const raster &pix, const rect<int32_t> &r, const uint8_t *ptr,
1466 int32_t stride) {

1468 const auto r_h = static_cast<size_t>(r.h < 0 ? 0 : r.h);
1469 for (size_t y = 0; y < r_h; y++) {
1470 float err_r = 0;
1471 float err_g = 0;
1472 float err_b = 0;
1473 for (size_t x = 0; x < r_w; x++) {
1474 const uint8_t *src = &ptr[y * static_cast<size_t>(stride) + x * 4];
1475 const float Rl = hidden::srgb_to_linear(static_cast<float>(src[0]) * (1.0f / 255.0f)) + err_r;
1476 const float Gl = hidden::srgb_to_linear(static_cast<float>(src[1]) * (1.0f / 255.0f)) + err_g;
1477 const float Bl = hidden::srgb_to_linear(static_cast<float>(src[2]) * (1.0f / 255.0f)) + err_b;
1478 if constexpr (BITS == 1) {
1479 const uint8_t n = (Rl * 2 + Gl * 3 + Bl * 1) > 3.0f ? 1 : 0;
1480 plot(pix, x + static_cast<size_t>(r.x), y + static_cast<size_t>(r.y), n);
1481 const float c = 0.75;
1482 err_r = std::clamp(Rl - ((n != uint8_t{0}) ? 1.0f : 0.0f), -c, c);
1483 err_g = std::clamp(Gl - ((n != uint8_t{0}) ? 1.0f : 0.0f), -c, c);
1484 err_b = std::clamp(Bl - ((n != uint8_t{0}) ? 1.0f : 0.0f), -c, c);
1485 } else {
1486 const uint8_t n = quant.nearest_linear(Rl, Gl, Bl);

3158 [[nodiscard]] constexpr uint8_t convert_byte(const convert_area &area, size_t line, size_t offset) const {
3159 if constexpr (dst_format == STRAIGHT_THROUGH) {
3160 return pix.data[(area.y + line) * pix.stride + convert_first_byte(area.x) + offset];
3161 } else if constexpr (dst_format == RGB565_8BIT_SERIAL) {
3162 const uint32_t col = pixel_rgb(area.x + offset / 2, area.y + line);
3163 if ((offset & 1) == 0) {

3165 }
3166 return static_cast<uint8_t>((((((col >> 8) & 0xff) >> 2) & 0x7) << 5) | (((col >> 16) & 0xff) >> 3));
3167 } else if constexpr (dst_format == RGB666_8BIT_SERIAL_1) {
3168 const uint32_t col = pixel_rgb(area.x + offset / 3, area.y + line);
3169 return static_cast<uint8_t>(((col >> (8 * (offset % 3))) & 0xff) >> 2);
3170 } else if constexpr (dst_format == RGB666_8BIT_SERIAL_2) {
3171 const uint32_t col = pixel_rgb(area.x + offset / 3, area.y + line);
3172 return static_cast<uint8_t>((((col >> (8 * (offset % 3))) & 0xff) >> 2) << 2);
3173 } else if constexpr (dst_format == MONO_1BIT_VERTICAL_PAGES) {
3174 static_assert(PF::bits_per_pixel == 1, "MONO_1BIT_VERTICAL_PAGES needs a 1-bit format.");

3177 const uint8_t *row = pix.data + (area.y + line * 8 + bit) * pix.stride;
3178 out |= static_cast<uint8_t>((PF::get_col(row, area.x + offset) & 1) << bit);
3179 }

3181 } else if constexpr (dst_format == MONO_1BIT_HORIZONTAL_LSB) {
3182 static_assert(PF::bits_per_pixel == 1, "MONO_1BIT_HORIZONTAL_LSB needs a 1-bit format.");

5075 // Decodes one code point of [str, end) and returns the position after it. Never reads at or past end: a
5076 // sequence cut off by the end decodes as 0.
5077 [[nodiscard]] static constexpr const char *next_utf32(const char *str, const char *end, uint32_t *utf32) {
5078 if (str == end) {
5079 *utf32 = 0;
5080 return str;
5081 }
5082 const uint32_t lead = static_cast<uint32_t>(str[0]) & 0xFF;
5083 const auto left = static_cast<size_t>(end - str);
5084 if (lead < 0x80) {
5085 *utf32 = lead;
5086 return str + 1;
5087 }
5088 if ((lead >> 5) == 0x06 && left >= 2) {
5089 *utf32 = ((lead & 0x1F) << 6) | (static_cast<uint32_t>(str[1]) & 0x3F);
5090 return str + 2;

5092 if ((lead >> 4) == 0x0E && left >= 3) {
5093 *utf32 = ((lead & 0x0F) << 12) | ((static_cast<uint32_t>(str[1]) & 0x3F) << 6) |
5094 (static_cast<uint32_t>(str[2]) & 0x3F);

5096 }

5098 *utf32 = ((lead & 0x07) << 18) | ((static_cast<uint32_t>(str[1]) & 0x3F) << 12) |

5470 } else if constexpr (ROTATION == DEGREE_270) {

5489 template <typename FONT, text_rotation ROTATION>
5490 constexpr void draw_char(int32_t x, int32_t y, const char_info<typename FONT::char_info_type> &ch, uint8_t col) {
5491 const auto dst = pix;
5492 constexpr int32_t ppb = FONT::mono ? 8 : 2; // glyph pixels per byte
5493 int32_t ch_data_off = static_cast<int32_t>(ch.y) * static_cast<int32_t>(FONT::glyph_bitmap_stride) +
5494 static_cast<int32_t>(ch.x) / ppb;
5495 std::array<float, 3> lin{};
5496 if constexpr (!FONT::mono) {
5497 lin = linear_color(col);
5498 }
5499
5500 auto limit_box = [&](int32_t &xmin, int32_t &ymin, int32_t &xmax, int32_t &ymax) {
5501 ymin = std::max(ymin, int32_t{0});
5502 ymax = std::min(ymax, static_cast<int32_t>(pix.height) - int32_t{1});
5503 xmin = std::max(xmin, int32_t{0});
5504 xmax = std::min(xmax, static_cast<int32_t>(pix.width) - int32_t{1});
5505 };
5506
5507 // Draws pixel x_off of the current glyph line at xx/yy.
5508 auto put = [&](int32_t xx, int32_t yy, int32_t x_off) {
5509 const int32_t bit_index = (ppb - 1 - (x_off % ppb)) * (8 / ppb);
5510 const auto byte_index = static_cast<size_t>(ch_data_off + x_off / ppb);
5511 if (byte_index < (FONT::glyph_bitmap_stride * FONT::glyph_bitmap_height)) {
5512 constexpr uint8_t full = FONT::mono ? 1 : 0xF;
5513 const auto a = static_cast<uint8_t>((FONT::glyph_bitmap[byte_index] >> bit_index) & full);
5514 if (a == full) {
5515 plot_unsafe(dst, xx, yy, col);
5516 } else if constexpr (!FONT::mono) {
5517 if (a != 0) {
5518 compose_unsafe(dst, xx, yy, hidden::a2al_4bit[a], lin[0], lin[1], lin[2]);
5519 }
5520 }
5521 }
5522 };
5523
5524 if constexpr (ROTATION == DEGREE_0) {
5525 if (check_not_in_bounds(x + ch.xoffset, y + ch.yoffset, ch.width + 1, ch.height + 1)) {
5526 return;
5527 }
5528
5529 x += ch.xoffset;
5530 y += ch.yoffset;
5531 int32_t x2 = x + static_cast<int32_t>(ch.width);
5532 int32_t y2 = y + static_cast<int32_t>(ch.height);
5533 limit_box(x, y, x2, y2);
5534 for (int32_t yy = y; yy < y2; yy++) {
5535 for (int32_t xx = x; xx < x2; xx++) {
5536 put(xx, yy, (xx - x) + ch.x % ppb);
5537 }
5538 ch_data_off += FONT::glyph_bitmap_stride;
5539 }
5540 } else if constexpr (ROTATION == DEGREE_180) {
5541 if (check_not_in_bounds(x - ch.xoffset - ch.width, y - FONT::ascent, ch.width + 1, ch.height + 1)) {
5542 return;
5543 }
5544
5545 x -= ch.xoffset;
5546 y -= FONT::ascent;
5547 int32_t x2 = x - static_cast<int32_t>(ch.width);
5548 int32_t y2 = y + static_cast<int32_t>(ch.height);
5549 limit_box(x2, y, x, y2);
5550 for (int32_t yy = y2 - 1; yy >= y; yy--) {
5551 for (int32_t xx = x2; xx < x; xx++) {
5552 put(xx, yy, (ch.width - (xx - x2) - 1) + ch.x % ppb);
5553 }
5554 ch_data_off += FONT::glyph_bitmap_stride;
5555 }
5556 } else if constexpr (ROTATION == DEGREE_90) {
5557 if (check_not_in_bounds(x - FONT::ascent, y, ch.height + 1, ch.width + 1)) {
5558 return;
5559 }
5560
5561 x -= FONT::ascent;
5562 y += ch.xoffset;
5563 int32_t x2 = x + static_cast<int32_t>(ch.height);
5564 int32_t y2 = y + static_cast<int32_t>(ch.width);
5565 limit_box(x, y, x2, y2);
5566 for (int32_t xx = x2 - 1; xx >= x; xx--) {
5567 for (int32_t yy = y; yy < y2; yy++) {
5568 put(xx, yy, (yy - y) + ch.x % ppb);
5569 }
5570 ch_data_off += FONT::glyph_bitmap_stride;
5571 }
5572 } else if constexpr (ROTATION == DEGREE_270) {
5573 if (check_not_in_bounds(x + ch.yoffset, y - ch.xoffset - ch.width, ch.height + 1, ch.width + 1)) {
5574 return;
5575 }
5576
5577 x += ch.yoffset;
5578 y -= ch.xoffset;
5579 int32_t x2 = x + static_cast<int32_t>(ch.height);
5580 int32_t y2 = y - static_cast<int32_t>(ch.width);
5581 limit_box(x, y2, x2, y);
5582 for (int32_t xx = x; xx < x2; xx++) {
5583 for (int32_t yy = y2; yy < y; yy++) {
5584 put(xx, yy, (ch.width - (yy - y2) - 1) + ch.x % ppb);
5585 }
5586 ch_data_off += FONT::glyph_bitmap_stride;
5587 }
5588 }
5589 }
